CMAKE_MINIMUM_REQUIRED(VERSION 3.10)
PROJECT(AT_TEST)

# Set device
INCLUDE(Barcode.cmake)
IF (${TARGET_PLATFORM} MATCHES "imx8mini")
    SET(DEVICE VS800)
    LOAD_BARCODE("/opt/libs_aarch64/Barcode_mini8_0.24.0")
ELSEIF (${TARGET_PLATFORM} MATCHES "imx8plus")
    SET(DEVICE VS1000Pro)
    LOAD_BARCODE("/opt/libs_aarch64/Barcode_8plus_dl_0.25.3")
ENDIF ()

# Build AT_TEST
SET(INC_DIR "./drivers/${DEVICE}/" "./")
FILE(GLOB AT_TEST_SRC "./main.cpp" "./cap_image.cpp")
FILE(GLOB VCAP_SRC "./drivers/${DEVICE}/*.c*")
ADD_EXECUTABLE(AT_TEST ${AT_TEST_SRC} ${VCAP_SRC})

# Determine whether to load barcode and link it
IF (${TARGET_SERIES} MATCHES "VS")
    TARGET_INCLUDE_DIRECTORIES(AT_TEST PUBLIC ${INC_DIR} ${Barcode_INCLUDE_DIRS})
    TARGET_LINK_LIBRARIES(AT_TEST PUBLIC AT ${OpenCV_LIBS} ${Barcode_LIBRARIES})
ELSEIF (${TARGET_SERIES} MATCHES "VN")
    TARGET_INCLUDE_DIRECTORIES(AT_TEST PUBLIC ${INC_DIR})
    TARGET_LINK_LIBRARIES(AT_TEST PUBLIC AT ${OpenCV_LIBS})
ENDIF ()

# Install
INSTALL(TARGETS AT_TEST
        DESTINATION bin)
